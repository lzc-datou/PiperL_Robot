cmake_minimum_required(VERSION 3.16)
project(piper_multi_point_rviz)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/TaskCommand.srv")
rosidl_get_typesupport_target(typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
add_library(piper_task_panel SHARED task_panel.cpp task_panel.hpp motion_controls.cpp motion_controls.hpp scene_editor.cpp scene_editor.hpp timed_displays.cpp ../../common/rviz_theme.qrc)
ament_target_dependencies(piper_task_panel rclcpp rviz_common pluginlib std_msgs)
target_link_libraries(piper_task_panel Qt5::Widgets "${typesupport_target}")
if(rviz_common_VERSION VERSION_EQUAL "14.1.23")
  target_compile_definitions(piper_task_panel PRIVATE PIPER_RVIZ_NANOSECOND_DELTA)
endif()
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS piper_task_panel LIBRARY DESTINATION lib)
ament_export_dependencies(rosidl_default_runtime)
ament_package()
